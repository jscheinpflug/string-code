#+interface:

Presets are the physics-facing constructors for common worldsheet CFT data.
They sit between [[id:20260605T090102.250727][Theory]] and
[[id:20260606T150002.000000][CFT Kernel]].  A user asks for a free boson, a \(bc\)
system, a product theory, or a boundary extension, and receives a generated
type with a small operator namespace and named correlator configs.

The implementation is split by audit target:

- [[id:20260606T183100.000000][Preset Shared Runtime]] defines compact handles,
  local-operator input, Wick templates, zero-mode templates, and the streaming
  correlator boundary.
- [[id:20260606T183200.000000][Free Boson Preset]] defines the free-boson bulk and
  Neumann/Dirichlet disk data.
- [[id:20260606T183300.000000][bc Preset]] defines the \(bc\) sphere and disk data.
- [[id:20260608T090000.000000][Free Fermion Preset]] defines the NS free fermion
  on the sphere.
- [[id:20260609T233500.000000][Primitive Pair Kernels]] exposes the primitive
  free-field pair-kernel inventory used by clients that need local contraction
  type data without the full correlator engine.
- [[id:20260608T100000.000000][eta-xi Preset]] defines the \(\eta,\xi\) system
  on the sphere and records its torus draft metadata.
- [[id:20260606T183400.000000][Preset Composition]] joins bulk sectors and boundary
  extensions without introducing a new runtime operator representation.
- [[id:20260609T004000.000000][Lisp Preset Compiler]] and the Lisp preset nodes
  declare descriptor fixtures that are translated to the generated Zig ABI
  module for REPL use.

This node is an internal preset bridge used by [[id:20260603T063945.054265][CFT-Code]].
It is not the user API.  It re-exports only the declarations the root CFT API
needs to build compact family objects such as =FreeBoson= and =Bc=.  Raw
Wick-rule data, correlator config records, and runtime resolver types stay in
[[id:20260606T183100.000000][Preset Shared Runtime]] for implementation modules.

fn labelSymbol(value: anytype) !u32 {
    return switch (value) {
        .symbol => |symbol| symbol,
        else => error.UnexpectedLabel,
    };
}

fn slotOf(raw: u32, expected: []const u32) !u6 {
    for (expected, 0..) |candidate, index| {
        if (raw == candidate) return @intCast(index);
    }
    return error.UnexpectedSlot;
}

fn unorderedPairBit(left: u6, right: u6) u64 {
    const a = if (left < right) left else right;
    const b = if (left < right) right else left;
    return @as(u64, 1) << @intCast(a * 8 + b);
}

fn pairMask(left_raw: u32, right_raw: u32, expected: []const u32) !u64 {
    return unorderedPairBit(try slotOf(left_raw, expected), try slotOf(right_raw, expected));
}

fn allFourPairMask() u64 {
    return unorderedPairBit(0, 1) | unorderedPairBit(0, 2) | unorderedPairBit(0, 3) |
        unorderedPairBit(1, 2) | unorderedPairBit(1, 3) | unorderedPairBit(2, 3);
}

const ParitySink = struct {
    hash: u64 = 14695981039346656037,
    branches: usize = 0,
    terms: usize = 0,
    scalars: usize = 0,
    coordinates: usize = 0,
    tensors: usize = 0,
    residuals: usize = 0,
    term_ends: usize = 0,
    zero_modes: usize = 0,
    bases: usize = 0,

    fn mix(self: *@This(), value: u64) void {
        self.hash = (self.hash ^ value) *% 1099511628211;
    }

    fn mixI64(self: *@This(), value: i64) void {
        self.mix(@as(u64, @bitCast(value)));
    }

    fn mixBytes(self: *@This(), bytes: []const u8) void {
        self.mix(bytes.len);
        for (bytes) |byte| self.mix(byte);
    }

    fn mixOptionalU8(self: *@This(), value: ?u8) void {
        if (value) |unwrapped| {
            self.mix(1);
            self.mix(unwrapped);
        } else {
            self.mix(0);
        }
    }

    fn mixRuleScalar(self: *@This(), scalar: anytype) !void {
        switch (scalar) {
            .one => self.mix(10),
            .rational => |value| {
                self.mix(11);
                self.mixI64(value.numerator);
                self.mixI64(value.denominator);
            },
            .monomial => |value| {
                self.mix(12);
                self.mixI64(value.rational.numerator);
                self.mixI64(value.rational.denominator);
                self.mix(value.imaginary_power);
                self.mix(value.atom orelse 0);
                self.mixI64(value.atom_power);
            },
        }
    }

    fn mixLabel(self: *@This(), label: anytype) !void {
        switch (label) {
            .integer => |value| {
                self.mix(20);
                self.mixI64(value);
            },
            .rational => |value| {
                self.mix(21);
                self.mixI64(value.numerator);
                self.mixI64(value.denominator);
            },
            .tensor => |value| {
                self.mix(22);
                self.mix(value);
            },
            .symbol => |value| {
                self.mix(23);
                self.mix(value);
            },
        }
    }

    fn mixCoordinateDifference(self: *@This(), coordinate: anytype) void {
        self.mix(coordinate.left);
        self.mix(coordinate.right);
    }

    fn mixInsertion(self: *@This(), insertion: anytype) void {
        switch (insertion) {
            .single => |item| {
                self.mix(60);
                self.mix(item.position.raw);
                self.mix(item.derivatives);
            },
            .pair => |item| {
                self.mix(61);
                self.mix(item.holomorphic_position.raw);
                self.mix(item.antiholomorphic_position.raw);
                self.mix(item.holomorphic_derivatives);
                self.mix(item.antiholomorphic_derivatives);
            },
        }
    }

    /// emitWickBranchSign records the generic Wick branch parity.
    pub fn emitWickBranchSign(self: *@This(), sign: i8) !void {
        self.branches += 1;
        self.mix(100);
        self.mixI64(sign);
    }

    /// emitWickTermStart records one primitive Wick term boundary.
    pub fn emitWickTermStart(self: *@This(), event: anytype) !void {
        self.terms += 1;
        self.mix(101);
        self.mix(event.left_index);
        self.mix(event.right_index);
        self.mix(event.term_index);
    }

    /// emitWickScalar records a scalar multiplier.
    pub fn emitWickScalar(self: *@This(), factor: anytype) !void {
        self.scalars += 1;
        self.mix(102);
        switch (factor) {
            .value => |value| try self.mixRuleScalar(value),
            .label_bilinear_phase => |value| {
                self.mix(13);
                try self.mixLabel(value.left);
                try self.mixLabel(value.right);
                self.mixBytes(value.form);
            },
            .cocycle => |value| {
                self.mix(14);
                self.mixBytes(value.table);
                try self.mixLabel(value.left);
                try self.mixLabel(value.right);
            },
            .spin_structure => |name| {
                self.mix(15);
                self.mixBytes(name);
            },
        }
    }

    /// emitWickCoordinate records a coordinate multiplier.
    pub fn emitWickCoordinate(self: *@This(), factor: anytype) !void {
        self.coordinates += 1;
        self.mix(103);
        self.mixI64(factor.scalar.numerator);
        self.mixI64(factor.scalar.denominator);
        switch (factor.kernel) {
            .difference_power => |power| {
                self.mix(30);
                self.mixCoordinateDifference(power.coordinate);
                self.mixI64(power.exponent);
            },
            .point_power => |power| {
                self.mix(31);
                self.mix(power.point);
                self.mixI64(power.exponent);
            },
            .logarithm => |logarithm| {
                self.mix(32);
                self.mixCoordinateDifference(logarithm);
            },
            .green_kernel => |kernel| {
                self.mix(33);
                self.mixBytes(kernel.name);
                self.mixCoordinateDifference(kernel.coordinate);
                self.mix(kernel.left_derivatives);
                self.mix(kernel.right_derivatives);
            },
            .green_exponential => |coordinate| {
                self.mix(34);
                self.mixCoordinateDifference(coordinate);
            },
        }
    }

    /// emitWickTensor records a tensor multiplier.
    pub fn emitWickTensor(self: *@This(), factor: anytype) !void {
        self.tensors += 1;
        self.mix(104);
        switch (factor) {
            .none => self.mix(40),
            .metric => |metric| {
                self.mix(41);
                try self.mixLabel(metric.left);
                try self.mixLabel(metric.right);
            },
            .momentum_pair => |pair| {
                self.mix(42);
                try self.mixLabel(pair.left);
                try self.mixLabel(pair.right);
            },
            else => return error.UnexpectedTensorFactor,
        }
    }

    /// emitWickAction rejects parity cases that unexpectedly stream actions.
    pub fn emitWickAction(_: *@This(), _: anytype) !void {
        return error.UnexpectedActionFactor;
    }

    /// emitWickResidualOperator records surviving input operators.
    pub fn emitWickResidualOperator(self: *@This(), residual: anytype) !void {
        self.residuals += 1;
        self.mix(105);
        self.mix(residual.input_index);
        self.mix(residual.operator.kind);
        self.mixInsertion(residual.operator.insertion);
    }

    /// emitWickTermEnd records a primitive term boundary.
    pub fn emitWickTermEnd(self: *@This()) !void {
        self.term_ends += 1;
        self.mix(106);
    }

    /// emitZeroModeFactor records zero-mode base-case data.
    pub fn emitZeroModeFactor(self: *@This(), factor: anytype) !void {
        self.zero_modes += 1;
        self.mix(107);
        switch (factor) {
            .bc_top_form => |top| {
                self.mix(50);
                self.mix(@intFromEnum(top.support));
                try self.mixRuleScalar(top.normalization);
                for (top.jets) |jet| {
                    self.mix(jet.coordinate.raw);
                    self.mix(jet.derivative_order);
                }
            },
            .eta_xi_zero_mode => |zero| {
                self.mix(51);
                self.mix(@intFromEnum(zero.support));
                try self.mixRuleScalar(zero.normalization);
                self.mix(zero.xi.coordinate.raw);
                self.mix(zero.xi.derivative_order);
            },
            .momentum_delta => |delta| {
                self.mix(52);
                try self.mixRuleScalar(delta.scalar);
                for (delta.scalar_factors) |scalar| try self.mixRuleScalar(scalar);
                for (delta.momenta) |momentum| try self.mixLabel(momentum);
            },
            else => return error.UnexpectedZeroModeFactor,
        }
    }

    /// emitZeroModeBaseEnd records an accepted residual base case.
    pub fn emitZeroModeBaseEnd(self: *@This()) !void {
        self.bases += 1;
        self.mix(108);
    }
};

const BasisDigest = struct {
    count: usize = 0,
    hash_xor: u64 = 0,
    hash_sum: u64 = 0,
    hash_square_sum: u64 = 0,
    mode_count_sum: usize = 0,
    weight_sum: i64 = 0,
    level_sum: u64 = 0,

    fn fnv(hash: u64, value: u64) u64 {
        return (hash ^ value) *% 1099511628211;
    }

    fn fnvI64(hash: u64, value: i64) u64 {
        return fnv(hash, @as(u64, @bitCast(value)));
    }

    fn candidateHash(candidate: anytype) u64 {
        var hash: u64 = 14695981039346656037;
        hash = fnv(hash, candidate.presentation);
        hash = fnv(hash, candidate.seed);
        hash = fnv(hash, candidate.seed_body);
        hash = fnvI64(hash, candidate.base_weight_ticks);
        hash = fnvI64(hash, candidate.weight_ticks);
        hash = fnv(hash, candidate.level_ticks);
        for (candidate.base_component_weight_ticks) |value| hash = fnvI64(hash, value);
        for (candidate.base_quantum_values) |value| hash = fnvI64(hash, value);
        for (candidate.component_weight_ticks) |value| hash = fnvI64(hash, value);
        for (candidate.quantum_values) |value| hash = fnvI64(hash, value);
        for (candidate.modes) |mode| {
            hash = fnv(hash, mode.id);
            hash = fnv(hash, mode.component);
            hash = fnv(hash, mode.label);
            hash = fnv(hash, mode.weight_ticks);
            hash = fnvI64(hash, mode.base_weight_ticks);
            hash = fnv(hash, mode.derivative_step_ticks);
            hash = fnv(hash, @intFromEnum(mode.statistics));
            hash = fnv(hash, mode.body);
        }
        return hash;
    }

    /// emitBasisState records one compact basis candidate.
    pub fn emitBasisState(self: *@This(), candidate: anytype) !void {
        const hash = candidateHash(candidate);
        self.count += 1;
        self.hash_xor ^= hash;
        self.hash_sum +%= hash;
        self.hash_square_sum +%= hash *% hash;
        self.mode_count_sum += candidate.modes.len;
        self.weight_sum += candidate.weight_ticks;
        self.level_sum += candidate.level_ticks;
    }
};

fn expectBasisDigestEqual(expected: BasisDigest, actual: BasisDigest) !void {

    try testing.expectEqual(expected.hash_xor, actual.hash_xor);
    try testing.expectEqual(expected.hash_sum, actual.hash_sum);

fn cfgHasField(comptime cfg: anytype, comptime name: []const u8) bool {
    return switch (@typeInfo(@TypeOf(cfg))) {
        .@"struct" => @hasField(@TypeOf(cfg), name),
        else => false,
    };
}

fn cfgDimension(comptime cfg: anytype, comptime constructor: []const u8) u16 {
    if (!cfgHasField(cfg, "dimension")) {
        @compileError(constructor ++ " requires .{ .dimension = 10 }");
    }
    return cfg.dimension;
}

fn cfgIncludeAntiholomorphic(comptime cfg: anytype) bool {
    if (cfgHasField(cfg, "include_antiholomorphic_copy")) return cfg.include_antiholomorphic_copy;
    return false;
}

fn publicFreeBosonMake(comptime cfg: anytype) type {
    const dimension = cfgDimension(cfg, "cft.FreeBoson.make");
    if (dimension != 10) {
        @compileError("cft.FreeBoson.make is generated only for dimension=10; instantiate a Lisp descriptor for another dimension");
    }
    return freeBoson(.{ .dimension = 10 });
}

fn publicBcSphere(comptime cfg: anytype) type {
    return bcSphere(.{ .include_antiholomorphic_copy = cfgIncludeAntiholomorphic(cfg) });
}

fn publicFreeFermionSphere(comptime cfg: anytype) type {

test "free fermion sphere uses generic fermionic Wick signs" {
    const testing = @import("std").testing;

    const Psi = freeFermionSphere(.{ .dimension = 10, .include_antiholomorphic_copy = false });

    var local = try Psi.local(testing.allocator);
    defer local.deinit();

    const mu = try local.index("mu");
    const nu = try local.index("nu");
    const rho = try local.index("rho");
    const sigma = try local.index("sigma");
    const z1 = try local.coord("z1");
    const z2 = try local.coord("z2");
    const z3 = try local.coord("z3");
    const z4 = try local.coord("z4");
    const ops = try local.ops(.{
        try Psi.op.psi(&local, mu, 0, z1),
        try Psi.op.psi(&local, nu, 0, z2),
        try Psi.op.psi(&local, rho, 0, z3),
        try Psi.op.psi(&local, sigma, 0, z4),
    });

    const Sink = struct {
        pub const emitWickScalar = noopWickScalar;
        pub const emitWickAction = noopWickAction;
        pub const emitWickTermEnd = noopWickTermEnd;
        pub const emitZeroModeFactor = noopZeroModeFactor;
        pub const emitZeroModeBaseEnd = noopZeroModeBaseEnd;

        coords: [4]u32,
        labels: [4]u32,
        branch_count: usize = 0,
        primitive_count: usize = 0,
        negative_branch_count: usize = 0,
        pending_negative: bool = false,
        coordinate_mask: u64 = 0,
        tensor_mask: u64 = 0,

        /// emitWickBranchSign records signs supplied by the generic branch walker.
        pub fn emitWickBranchSign(self: *@This(), sign: i8) !void {
            if (sign < 0) self.pending_negative = true;
        }

        /// emitWickTermStart records the first primitive term of each Wick branch.
        pub fn emitWickTermStart(self: *@This(), _: anytype) !void {
            self.primitive_count += 1;
            if ((self.primitive_count % 2) == 1) {
                self.branch_count += 1;
                if (self.pending_negative) {
                    self.negative_branch_count += 1;
                    self.pending_negative = false;
                }
            }
        }

        /// emitWickCoordinate records every primitive fermion pole.
        pub fn emitWickCoordinate(self: *@This(), factor: anytype) !void {
            switch (factor.kernel) {
                .difference_power => |power| {
                    if (power.exponent != -1) return error.UnexpectedCoordinateFactor;
                    self.coordinate_mask |= try pairMask(power.coordinate.left, power.coordinate.right, &self.coords);
                },
                else => return error.UnexpectedCoordinateFactor,
            }
        }

        /// emitWickTensor records every metric pairing.
        pub fn emitWickTensor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .metric => |metric| self.tensor_mask |= try pairMask(try labelSymbol(metric.left), try labelSymbol(metric.right), &self.labels),
                else => return error.UnexpectedTensorFactor,
            }
        }
    };

    var sink = Sink{
        .coords = .{ @intFromEnum(z1), @intFromEnum(z2), @intFromEnum(z3), @intFromEnum(z4) },
        .labels = .{ @intFromEnum(mu), @intFromEnum(nu), @intFromEnum(rho), @intFromEnum(sigma) },
    };
    try Psi.correlator(&Psi.config.sphere, ops, &sink);
    try testing.expectEqual(@as(usize, 3), sink.branch_count);
    try testing.expectEqual(@as(usize, 6), sink.primitive_count);
    try testing.expectEqual(@as(usize, 1), sink.negative_branch_count);
    try testing.expectEqual(allFourPairMask(), sink.coordinate_mask);
    try testing.expectEqual(allFourPairMask(), sink.tensor_mask);
    try testing.expect(!@hasDecl(@TypeOf(Psi.config.sphere), "fermion_kinds"));
    try testing.expect(!@hasDecl(@TypeOf(Psi.config.sphere), "pair_lookup"));
}

test "bc zero modes consume c jets by top-form saturation" {
    const testing = @import("std").testing;

    const Ghost = bcSphere(.{ .include_antiholomorphic_copy = false });

    var local = try Ghost.local(testing.allocator);
    defer local.deinit();

    const z1 = try local.coord("z1");
    const z2 = try local.coord("z2");
    const z3 = try local.coord("z3");
    const ops = try local.ops(.{
        try Ghost.op.c(&local, 0, z1),
        try Ghost.op.c(&local, 1, z2),
        try Ghost.op.c(&local, 2, z3),
    });

    const Sink = struct {
        pub const emitWickTermStart = noopWickTermStart;
        pub const emitWickScalar = noopWickScalar;
        pub const emitWickCoordinate = noopWickCoordinate;
        pub const emitWickTensor = noopWickTensor;
        pub const emitWickAction = noopWickAction;
        pub const emitWickTermEnd = noopWickTermEnd;

        coords: [3]u32,
        factor_count: usize = 0,
        base_end_count: usize = 0,
        derivative_sum: u16 = 0,
        coordinate_mask: u8 = 0,
        derivative_mask: u8 = 0,

        /// emitZeroModeFactor records the top-form factor emitted by the base case.
        pub fn emitZeroModeFactor(self: *@This(), factor: anytype) !void {
            self.factor_count += 1;
            switch (factor) {
                .bc_top_form => |top| {
                    if (top.support != .sphere_holomorphic) return error.UnexpectedBcSupport;
                    for (top.jets) |jet| {
                        self.coordinate_mask |= @as(u8, 1) << @intCast(try slotOf(jet.coordinate.raw, &self.coords));
                        self.derivative_mask |= @as(u8, 1) << @intCast(jet.derivative_order);
                        self.derivative_sum += jet.derivative_order;
                    }
                },
                else => return error.UnexpectedZeroModeFactor,
            }
        }

        /// emitZeroModeBaseEnd records that every residual insertion was consumed.
        pub fn emitZeroModeBaseEnd(self: *@This()) !void {
            self.base_end_count += 1;
        }
    };

    var sink = Sink{ .coords = .{ @intFromEnum(z1), @intFromEnum(z2), @intFromEnum(z3) } };
    try Ghost.correlator(&Ghost.config.sphere, ops, &sink);
    try testing.expectEqual(@as(usize, 1), sink.factor_count);
    try testing.expectEqual(@as(usize, 1), sink.base_end_count);
    try testing.expectEqual(@as(u16, 3), sink.derivative_sum);
    try testing.expectEqual(@as(u8, 0b111), sink.coordinate_mask);
    try testing.expectEqual(@as(u8, 0b111), sink.derivative_mask);
}

test "free boson zero modes emit momentum delta and profile presentations" {
    const testing = @import("std").testing;

    const X = freeBoson(.{ .dimension = 10 });

    var local = try X.local(testing.allocator);
    defer local.deinit();

    const k = try local.momentum("k");
    const z = try local.coord("z");
    const zbar = try local.coord("zbar");
    const f0 = try local.targetFunction("f");
    const fhat = try local.fourierTransform("fhat");
    const coeff = try local.profileCoefficient("a");
    const point = try local.targetPoint("x0");
    const position_profile = X.profile.positionSpace(f0);
    const fourier_profile = X.profile.fourier(fhat);
    const polynomial_profile = X.profile.polynomialRnc(point, .{.{ coeff, 2 }});
    const ops = try local.ops(.{
        try X.op.expX(&local, k, z, zbar),
        try X.op.profile(&local, position_profile, z, zbar),
        try X.op.profile(&local, fourier_profile, z, zbar),
        try X.op.profile(&local, polynomial_profile, z, zbar),
    });

    const Sink = struct {
        pub const emitWickTermStart = noopWickTermStart;
        pub const emitWickScalar = noopWickScalar;
        pub const emitWickCoordinate = noopWickCoordinate;
        pub const emitWickTensor = noopWickTensor;
        pub const emitWickAction = noopWickAction;
        pub const emitWickTermEnd = noopWickTermEnd;

        delta_count: usize = 0,
        gaussian_count: usize = 0,
        fourier_count: usize = 0,
        polynomial_count: usize = 0,
        base_end_count: usize = 0,
        saw_general_normalization: bool = false,

        /// emitZeroModeFactor records each free-boson zero-mode factor by presentation.
        pub fn emitZeroModeFactor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .momentum_delta => |delta| {
                    self.delta_count += 1;
                    if (delta.scalar_factors.len != 0) return error.UnexpectedZeroModeNormalization;
                    switch (delta.scalar) {
                        .monomial => |monomial| {
                            if (monomial.rational.numerator != 1024 or monomial.rational.denominator != 1) return error.UnexpectedZeroModeNormalization;
                            if (monomial.imaginary_power != 0) return error.UnexpectedZeroModeNormalization;
                            if (monomial.atom != scalars.atom("free_boson", "pi")) return error.UnexpectedZeroModeNormalization;
                            if (monomial.atom_power != 10) return error.UnexpectedZeroModeNormalization;
                            self.saw_general_normalization = true;
                        },
                        else => return error.UnexpectedZeroModeNormalization,
                    }
                    if (delta.projector != null or delta.momenta.len != 1) return error.UnexpectedMomentumDelta;
                },
                .profile_gaussian_differential => self.gaussian_count += 1,
                .profile_fourier_integral => self.fourier_count += 1,
                .profile_polynomial_integral => self.polynomial_count += 1,
                else => return error.UnexpectedZeroModeFactor,
            }
        }

        /// emitZeroModeBaseEnd records that all profile and plane-wave residuals were consumed.
        pub fn emitZeroModeBaseEnd(self: *@This()) !void {
            self.base_end_count += 1;
        }
    };

    var sink = Sink{};
    try X.correlator(&X.config.sphere, ops, &sink);
    try testing.expectEqual(@as(usize, 1), sink.delta_count);
    try testing.expect(sink.saw_general_normalization);
    try testing.expectEqual(@as(usize, 1), sink.gaussian_count);
    try testing.expectEqual(@as(usize, 1), sink.fourier_count);
    try testing.expectEqual(@as(usize, 1), sink.polynomial_count);
    try testing.expectEqual(@as(usize, 1), sink.base_end_count);
}

test "disk free boson zero modes emit Neumann delta and Dirichlet phase" {
    const testing = @import("std").testing;

    const X = freeBoson(.{ .dimension = 10 });
    const XBoundary = freeBosonBoundary(.{
        .neumann = X.target.subspace(&.{ 0, 1, 2, 3 }),
        .dirichlet = X.target.complement(&.{ 0, 1, 2, 3 }),
        .dirichlet_position = X.target.point("x0"),
    });
    const Disk = boundary(product(.{X}), .{XBoundary});

    var local = try Disk.local(testing.allocator);
    defer local.deinit();

    const k = try local.momentum("k");
    const y = try local.boundaryCoord("y");
    const ops = try local.ops(.{
        try @TypeOf(XBoundary).op.expXBoundary(&local, k, y),
    });

    const Sink = struct {
        pub const emitWickTermStart = noopWickTermStart;
        pub const emitWickScalar = noopWickScalar;
        pub const emitWickCoordinate = noopWickCoordinate;
        pub const emitWickTensor = noopWickTensor;
        pub const emitWickAction = noopWickAction;
        pub const emitWickTermEnd = noopWickTermEnd;

        delta_count: usize = 0,
        phase_count: usize = 0,
        base_end_count: usize = 0,
        saw_disk_scalar: bool = false,
        saw_disk_volume: bool = false,
        saw_projector: bool = false,
        saw_position: bool = false,

        /// emitZeroModeFactor records Neumann momentum conservation and Dirichlet phase data.
        pub fn emitZeroModeFactor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .momentum_delta => |delta| {
                    self.delta_count += 1;
                    self.saw_projector = delta.projector != null;
                    switch (delta.scalar) {
                        .monomial => |monomial| self.saw_disk_scalar = monomial.atom == scalars.atom("free_boson", "K_D2") and monomial.atom_power == 1,
                        else => return error.UnexpectedDiskNormalization,
                    }
                    if (delta.scalar_factors.len != 1) return error.UnexpectedDiskNormalization;
                    switch (delta.scalar_factors[0]) {
                        .monomial => |monomial| {
                            if (monomial.rational.numerator != 16 or monomial.rational.denominator != 1) return error.UnexpectedDiskNormalization;
                            if (monomial.imaginary_power != 0) return error.UnexpectedDiskNormalization;
                            if (monomial.atom != scalars.atom("free_boson", "pi")) return error.UnexpectedDiskNormalization;
                            if (monomial.atom_power != 4) return error.UnexpectedDiskNormalization;
                            self.saw_disk_volume = true;
                        },
                        else => return error.UnexpectedDiskNormalization,
                    }
                    if (delta.momenta.len != 1) return error.UnexpectedMomentumDelta;
                },
                .dirichlet_phase => |phase| {
                    self.phase_count += 1;
                    self.saw_position = phase.momenta.len == 1;
                    switch (phase.position) {
                        .target_point => {},
                        else => return error.UnexpectedDirichletPosition,
                    }
                },
                else => return error.UnexpectedZeroModeFactor,
            }
        }

        /// emitZeroModeBaseEnd records that the disk constant mode consumed the insertion.
        pub fn emitZeroModeBaseEnd(self: *@This()) !void {
            self.base_end_count += 1;
        }
    };

    var sink = Sink{};
    try Disk.correlator(&Disk.config.disk, ops, &sink);
    try testing.expectEqual(@as(usize, 1), sink.delta_count);
    try testing.expectEqual(@as(usize, 1), sink.phase_count);
    try testing.expect(sink.saw_disk_scalar);
    try testing.expect(sink.saw_disk_volume);
    try testing.expect(sink.saw_projector);
    try testing.expect(sink.saw_position);
    try testing.expectEqual(@as(usize, 1), sink.base_end_count);
}

    switch (scalars.rational(2, 4)) {
        .rational => |value| {
            try testing.expectEqual(@as(i64, 1), value.numerator);
            try testing.expectEqual(@as(i64, 2), value.denominator);
        },
        else => return error.UnexpectedScalarShape,
    }

    const Sink = struct {

        saw_coordinate: bool = false,

        /// emitWickAction rejects unexpected action factors.
        pub fn emitWickAction(_: *@This(), _: anytype) !void {
            return error.UnexpectedAction;
        }

        /// emitWickTermEnd accepts the completed primitive contraction.
        pub fn emitWickTermEnd(_: *@This()) !void {}

        /// emitZeroModeFactor accepts the empty free-boson constant mode.
        pub fn emitZeroModeFactor(_: *@This(), _: anytype) !void {}

        /// emitZeroModeBaseEnd accepts the full contraction base case.
        pub fn emitZeroModeBaseEnd(_: *@This()) !void {}
    };

    var sink = Sink{ .expected_alpha = preset_alpha };

        /// emitWickTermStart records one primitive dX contraction.

                        self.z1_points += 1;
                    } else if (power.point == self.z2) {
                        self.z2_points += 1;
                    } else {
                        return error.UnexpectedCoordinateFactor;
                    }
                    self.coordinates += 1;
                },
                else => return error.UnexpectedCoordinateFactor,
            }
        }

        pub fn emitWickTensor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .metric => self.tensors += 1,
                else => return error.UnexpectedTensorFactor,
            }
        }

        pub fn emitWickTermEnd(_: *@This()) !void {}

        pub fn emitZeroModeBaseEnd(self: *@This()) !void {
            self.base_cases += 1;
        }
    };

    var sink = Sink{ .z1 = @intFromEnum(z1), .z2 = @intFromEnum(z2) };
    try X.correlator(&X.config.sphere, ops, &sink);
    try testing.expectEqual(@as(usize, 4), sink.scalars);
    try testing.expectEqual(@as(usize, 4), sink.coordinates);
    try testing.expectEqual(@as(usize, 2), sink.z1_points);
    try testing.expectEqual(@as(usize, 2), sink.z2_points);
    try testing.expectEqual(@as(usize, 4), sink.tensors);
    try testing.expectEqual(@as(usize, 2), sink.base_cases);
}

test "free boson exponential pairing at infinity leaves momentum delta only" {
    const testing = @import("std").testing;

    const X = freeBoson(.{ .dimension = 10 });

    var local = try X.local(testing.allocator);
    defer local.deinit();

    const zero = try local.coord("zero");
    const inf = local.infinity();
    const k = try local.momentum("k");
    const p = try local.momentum("p");
    const ops = try local.ops(.{
        try X.op.expX(&local, k, inf, inf),
        try X.op.expX(&local, p, zero, zero),
    });

    const Sink = struct {
        pub const emitWickScalar = noopWickScalar;
        pub const emitWickAction = noopWickAction;

        coordinates: usize = 0,
        tensors: usize = 0,
        deltas: usize = 0,
        base_cases: usize = 0,

        /// emitWickTermStart accepts the Green-exponential primitive term.
        pub fn emitWickTermStart(_: *@This(), _: anytype) !void {}

        /// emitWickCoordinate rejects leftover Green-exponential coordinate factors.
        pub fn emitWickCoordinate(self: *@This(), _: anytype) !void {
            self.coordinates += 1;
            return error.UnexpectedCoordinateFactor;
        }

        /// emitWickTensor records the finite momentum-pair tensor.
        pub fn emitWickTensor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .momentum_pair => self.tensors += 1,
                else => return error.UnexpectedTensorFactor,
            }
        }

        /// emitWickResidualOperator accepts the residual exponentials.
        pub fn emitWickResidualOperator(_: *@This(), _: anytype) !void {}

        /// emitWickTermEnd accepts the primitive boundary.
        pub fn emitWickTermEnd(_: *@This()) !void {}

        /// emitZeroModeFactor records the momentum-conservation delta.
        pub fn emitZeroModeFactor(self: *@This(), factor: anytype) !void {
            switch (factor) {
                .momentum_delta => |delta| {
                    try testing.expectEqual(@as(usize, 2), delta.momenta.len);
                    self.deltas += 1;
                },
                else => return error.UnexpectedZeroModeFactor,
            }
        }

        /// emitZeroModeBaseEnd records one accepted base case.
        pub fn emitZeroModeBaseEnd(self: *@This()) !void {
            self.base_cases += 1;
        }
    };

    var sink = Sink{};
    try X.correlator(&X.config.sphere, ops, &sink);
    try testing.expectEqual(@as(usize, 0), sink.coordinates);
    try testing.expectEqual(@as(usize, 1), sink.tensors);
    try testing.expectEqual(@as(usize, 1), sink.deltas);
    try testing.expectEqual(@as(usize, 1), sink.base_cases);
}
