The \(\eta,\xi\) presets share fields but use different surface kernels.  The
sphere rule uses the rational pole; the torus rule names the prime-form
derivative kernel.

#+begin_src lisp
(in-package #:string-code.cft.presets)
#+end_src

* Sphere

#+begin_src lisp
(define-cft-preset eta-xi-sphere
  (:theory-id 2
   :kind-namespace eta_xi
   :basis (:presentation eta-xi-sphere
           :backend (:eta-xi)
           :tick-denominator 1
           :quantum-numbers ((eta-xi-number))
           :seed-bits ((0 :operator xi :weight 0))))
  (surface sphere :sphere :rational)
  (quantum-number eta-xi-number (:u1))
  (chiral-field eta :fermionic
         :weight (rational 1 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number 1)))
  (chiral-field xi :fermionic
         :zero-mode t
         :weight (rational 0 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number -1)))
  (wick (eta z) (xi w)
    (/ 1 (- z w)))
  (zero-mode (grassmann-count :field xi)))
#+end_src

* Full Copies

The full-copy sphere and torus fixtures are authored descriptors.  They keep the
legacy antiholomorphic option out of Zig preset internals.

#+begin_src lisp
(define-cft-preset eta-xi-sphere-full
  (:theory-id 7
   :kind-namespace eta_xi
   :basis (:presentation eta-xi-sphere
           :backend (:eta-xi)
           :tick-denominator 1
           :quantum-numbers ((eta-xi-number))
           :seed-bits ((0 :operator xi :weight 0))))
  (surface sphere :sphere :rational)
  (quantum-number eta-xi-number (:u1))
  (chiral-field eta :fermionic
         :weight (rational 1 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number 1)))
  (chiral-field xi :fermionic
         :zero-mode t
         :weight (rational 0 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number -1)))
  (anti-chiral-field etat :fermionic
         :weight (rational 1 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number 1)))
  (anti-chiral-field xit :fermionic
         :zero-mode t
         :weight (rational 0 1)
         :infinity primary
         :quantum-numbers ((eta-xi-number -1)))
  (wick (eta z) (xi w)
    (/ 1 (- z w)))
  (wick (etat zbar) (xit wbar)
    (/ 1 (- zbar wbar)))
  (zero-mode (grassmann-count :field xi))
  (zero-mode (grassmann-count :field xit)))
