const StoredFactor = struct {
    field: u16 = 0,
    derivative: u8 = 0,
    label_count: u8 = 0,
    labels: [1]u32 = .{0},
};

fn ExpressionCollector(comptime d: anytype, comptime capacity: usize) type {
    return struct {
        const Entry = struct {
            used: bool = false,
            scalar: Ope.Scalar = .one,
            coordinates: [32]Ope.CoordinateAtom = undefined,
            coordinate_count: usize = 0,
            tensors: [32]Ope.TensorAtom = undefined,
            tensor_count: usize = 0,
            actions: [16]Ope.ActionAtom = undefined,
            action_count: usize = 0,
            output: [32]StoredFactor = undefined,
            output_count: usize = 0,
            branch_level: u8 = 0,
        };

        entries: [capacity]Entry = [_]Entry{.{}} ** capacity,

        pub fn emitOpeTerm(self: *@This(), term: Ope.TermView) !void {
            const entry = try storedEntry(term);
            for (&self.entries) |*slot| {
                if (slot.used and sameStructure(slot.*, entry)) {
                    try addScalar(&slot.scalar, entry.scalar);
                    return;
                }
            }
            for (&self.entries) |*slot| {
                if (!slot.used) {
                    slot.* = entry;
                    slot.used = true;
                    return;
                }
            }
            return error.ContextTooSmall;
        }

        fn storedEntry(term: Ope.TermView) !Entry {
            if (term.coordinates.len > 32 or term.tensors.len > 32 or term.actions.len > 16 or term.output.len > 32) return error.ContextTooSmall;
            var entry = Entry{
                .used = true,
                .scalar = term.scalar,
                .coordinate_count = term.coordinates.len,
                .tensor_count = term.tensors.len,
                .action_count = term.actions.len,
                .output_count = term.output.len,
                .branch_level = term.branch_level,
            };
            @memcpy(entry.coordinates[0..term.coordinates.len], term.coordinates);
            @memcpy(entry.tensors[0..term.tensors.len], term.tensors);
            @memcpy(entry.actions[0..term.actions.len], term.actions);
            for (term.output, 0..) |factor, index| {
                if (factor.labels.len > 1) return error.ContextTooSmall;
                entry.output[index] = .{
                    .field = factor.field,
                    .derivative = factor.derivative,
                    .label_count = @intCast(factor.labels.len),
                    .labels = .{if (factor.labels.len == 0) 0 else factor.labels[0]},
                };
            }
            return entry;
        }

        fn sameStructure(left: Entry, right: Entry) bool {
            if (left.branch_level != right.branch_level or
                left.coordinate_count != right.coordinate_count or
                left.tensor_count != right.tensor_count or
                left.action_count != right.action_count or
                left.output_count != right.output_count)
            {
                return false;
            }
            for (left.coordinates[0..left.coordinate_count], right.coordinates[0..right.coordinate_count]) |a, b| if (!std.meta.eql(a, b)) return false;
            for (left.tensors[0..left.tensor_count], right.tensors[0..right.tensor_count]) |a, b| if (!std.meta.eql(a, b)) return false;
            for (left.actions[0..left.action_count], right.actions[0..right.action_count]) |a, b| if (!std.meta.eql(a, b)) return false;
            for (left.output[0..left.output_count], right.output[0..right.output_count]) |a, b| if (!std.meta.eql(a, b)) return false;
            return true;
        }

        pub fn render(self: *@This(), comptime case: CaseName, allocator: std.mem.Allocator) ![]const u8 {
            var bytes: std.ArrayList(u8) = .empty;
            var writer = TextWriter{ .bytes = &bytes, .allocator = allocator };
            var emitted = false;
            for (self.entries) |entry| {
                if (!entry.used or entry.scalar.numerator == 0) continue;
                const green = hasGreen(entry);
                const negative = entry.scalar.numerator < 0 and !green;
                if (emitted) {
                    try writer.writeAll(if (negative) " - " else " + ");
                } else if (negative) {
                    try writer.writeAll("-");
                }
                var scalar = entry.scalar;
                if (scalar.numerator < 0 and !green) scalar.numerator = -scalar.numerator;
                try renderEntry(&writer, case, scalar, entry);
                emitted = true;
            }
            if (!emitted) try writer.writeAll("0");
            return bytes.toOwnedSlice(allocator);
        }

        fn renderEntry(writer: *TextWriter, comptime case: CaseName, scalar: Ope.Scalar, entry: Entry) !void {
            const green = hasGreen(entry);
            const consumed_momentum = if (green) firstMomentumPair(entry) else null;
            const consumed_tensor = if (consumed_momentum) |index| entry.tensors[index] else null;
            var factors: usize = 0;
            if (!green and (!scalarIsOne(scalar) or multiplicativeFactorCount(entry, null) == 0)) {
                try renderProductSeparator(writer, &factors);
                try renderScalar(writer, scalar);
            }
            for (entry.coordinates[0..entry.coordinate_count]) |coordinate| {
                try renderProductSeparator(writer, &factors);
                try renderCoordinate(writer, scalar, coordinate, consumed_tensor);
            }
            for (entry.tensors[0..entry.tensor_count], 0..) |tensor, index| {
                if (consumed_momentum) |consumed| if (index == consumed) continue;
                try renderProductSeparator(writer, &factors);
                try renderTensor(writer, case, tensor);
            }
            if (entry.output_count != 0) {
                try renderProductSeparator(writer, &factors);
                try renderOutput(writer, case, entry.output[0..entry.output_count]);
            }
            if (factors == 0) try writer.writeAll("1");
        }

        fn multiplicativeFactorCount(entry: Entry, consumed_momentum: ?usize) usize {
            var count = entry.coordinate_count + entry.output_count;
            for (entry.tensors[0..entry.tensor_count], 0..) |_, index| {
                if (consumed_momentum) |consumed| if (index == consumed) continue;
                count += 1;
            }
            return count;
        }

        fn hasGreen(entry: Entry) bool {
            for (entry.coordinates[0..entry.coordinate_count]) |coordinate| {
                if (std.meta.activeTag(coordinate) == .green_exponential) return true;
            }
            return false;
        }

        fn firstMomentumPair(entry: Entry) ?usize {
            for (entry.tensors[0..entry.tensor_count], 0..) |tensor, index| {
                if (std.meta.activeTag(tensor) == .momentum_pair) return index;
            }
            return null;
        }

        fn renderProductSeparator(writer: *TextWriter, factors: *usize) !void {
            if (factors.* != 0) try writer.writeAll("*");
            factors.* += 1;
        }

        fn renderCoordinate(writer: *TextWriter, scalar: Ope.Scalar, coordinate: Ope.CoordinateAtom, consumed_tensor: ?Ope.TensorAtom) !void {
            switch (coordinate) {
                .difference_power => |item| try format(writer, "({s} - {s})^({})", .{
                    coordinateName(item.left_slot, .left),
                    coordinateName(item.right_slot, .right),
                    item.exponent,
                }),
                .green_exponential => |item| {
                    try format(writer, "({s} - {s})^(", .{
                        coordinateName(item.left_slot, .left),
                        coordinateName(item.right_slot, .right),
                    });
                    try renderGreenExponent(writer, scalar, consumed_tensor);
                    try writer.writeAll(")");
                },
                else => return error.UnsupportedRenderedOpeTerm,
            }
        }

        fn renderGreenExponent(writer: *TextWriter, scalar: Ope.Scalar, consumed_tensor: ?Ope.TensorAtom) !void {
            try writer.writeAll("(");
            try renderSignedScalar(writer, scalar);
            if (consumed_tensor) |tensor| {
                try writer.writeAll("*");
                try renderTensor(writer, .boson_exp_pair_w0, tensor);
            }
            try writer.writeAll(")");
        }

        fn renderTensor(writer: *TextWriter, comptime case: CaseName, tensor: Ope.TensorAtom) !void {
            switch (tensor) {
                .none => try writer.writeAll("1"),
                .metric => |item| try format(writer, "\\[Delta][{s}, {s}]", .{ vectorLabel(case, item.left), vectorLabel(case, item.right) }),
                .momentum_pair => |item| try format(writer, "dot[{s}, {s}]", .{ momentumLabel(item.left), momentumLabel(item.right) }),
                .momentum_index => |item| try format(writer, "{s}[{s}]", .{ momentumLabel(item.momentum), vectorLabel(case, item.index) }),
            }
        }

        fn renderOutput(writer: *TextWriter, comptime case: CaseName, output: []const StoredFactor) !void {
            try writer.writeAll("R[");
            for (output, 0..) |factor, index| {
                if (index != 0) try writer.writeAll(", ");
                try renderFactor(writer, case, factor);
            }
            try writer.writeAll("]");
        }

        fn renderFactor(writer: *TextWriter, comptime case: CaseName, factor: StoredFactor) !void {
            const symbol = d.symbols[d.fields[factor.field].symbol];
            if (std.mem.eql(u8, symbol, "expX")) {
                try format(writer, "expX[{s}, 0, 0]", .{momentumLabel(factor.labels[0])});
            } else if (std.mem.eql(u8, symbol, "psi")) {
                try format(writer, "\\[Psi][{s}, {}, 0]", .{ vectorLabel(case, factor.labels[0]), factor.derivative });
            } else if (std.mem.eql(u8, symbol, "dX")) {
                try format(writer, "dX[{s}, {}, 0]", .{ vectorLabel(case, factor.labels[0]), factor.derivative });
            } else {
                try format(writer, "{s}[{}, 0]", .{ symbol, factor.derivative });
            }
        }

        fn renderScalar(writer: *TextWriter, scalar: Ope.Scalar) !void {
            const numerator = if (scalar.numerator < 0) -scalar.numerator else scalar.numerator;
            if (scalar.denominator != 1) try writer.writeAll("(");
            var factors: usize = 0;
            if (numerator != 1 or (scalar.atom_count == 0 and scalar.imaginary_power == 0)) {
                try renderProductSeparator(writer, &factors);
                try format(writer, "{}", .{numerator});
            }
            if (scalar.imaginary_power != 0) {
                try renderProductSeparator(writer, &factors);
                try writer.writeAll(switch (scalar.imaginary_power) {
                    1 => "I",
                    2 => "-1",
                    3 => "-I",
                    else => unreachable,
                });
            }
            for (scalar.atoms[0..scalar.atom_count]) |atom| {
                try renderProductSeparator(writer, &factors);
                try renderScalarAtom(writer, atom);
            }
            if (scalar.denominator != 1) try format(writer, ")/{}", .{scalar.denominator});
        }

        fn renderSignedScalar(writer: *TextWriter, scalar: Ope.Scalar) !void {
            if (scalar.numerator < 0) try writer.writeAll("-");
            var abs_scalar = scalar;
            if (abs_scalar.numerator < 0) abs_scalar.numerator = -abs_scalar.numerator;
            try renderScalar(writer, abs_scalar);
        }

        fn renderScalarAtom(writer: *TextWriter, atom: Ope.Scalar.Atom) !void {
            const symbol = d.symbols[atom.symbol];
            if (std.mem.eql(u8, symbol, "alpha-prime")) {
                try writer.writeAll("\\[Alpha]p");
            } else {
                try writer.writeAll(symbol);
            }
            if (atom.power != 1) try format(writer, "^({})", .{atom.power});
        }

        fn scalarIsOne(scalar: Ope.Scalar) bool {
            return scalar.numerator == 1 and scalar.denominator == 1 and scalar.imaginary_power == 0 and scalar.atom_count == 0;
        }

        fn coordinateName(slot: anytype, side: anytype) []const u8 {
            return switch (slot) {
                .position, .holomorphic => if (side == .left) "z1" else "z2",
                .antiholomorphic => if (side == .left) "z1b" else "z2b",
            };
        }

        fn vectorLabel(comptime case: CaseName, label: u32) []const u8 {
            return switch (case) {
                .boson_dx_pair_w0 => if (label == 1) "mu" else "nu",
                else => switch (label) {
                    1 => "mu1",
                    2 => "mu2",
                    3 => "mu3",
                    4 => "mu4",
                    5 => "mu5",
                    6 => "mu6",
                    else => "mu",
                },
            };
        }

        fn momentumLabel(label: u32) []const u8 {
            return switch (label) {
                1 => "k",
                2 => "p",
                else => "q",
            };
        }

        const TextWriter = struct {
            bytes: *std.ArrayList(u8),
            allocator: std.mem.Allocator,

            pub fn writeAll(self: *@This(), data: []const u8) !void {
                try self.bytes.appendSlice(self.allocator, data);
            }
        };

        fn format(writer: *TextWriter, comptime fmt: []const u8, args: anytype) !void {
            var buffer: [256]u8 = undefined;
            const text = try std.fmt.bufPrint(&buffer, fmt, args);
            try writer.writeAll(text);
        }
    };
}

fn render(comptime case: CaseName, allocator: std.mem.Allocator, comptime d: anytype, comptime Kernel: type, left_terms: []const Ope.Monomial, right_terms: []const Ope.Monomial, projection: Ope.Projection) ![]const u8 {
    var sink = ExpressionCollector(d, 4096){};
    try Kernel.opeProjected(.{ .terms = left_terms }, .{ .terms = right_terms }, projection, &sink);
    return sink.render(case, allocator);
}

fn addScalar(left: *Ope.Scalar, right: Ope.Scalar) !void {
    if (left.imaginary_power != right.imaginary_power or left.atom_count != right.atom_count) return error.IncompatibleScalar;
    for (left.atoms[0..left.atom_count], right.atoms[0..right.atom_count]) |a, b| {
        if (a.symbol != b.symbol or a.power != b.power) return error.IncompatibleScalar;
    }
    const left_num = try std.math.mul(i64, left.numerator, right.denominator);
    const right_num = try std.math.mul(i64, right.numerator, left.denominator);
    left.numerator = try std.math.add(i64, left_num, right_num);
    left.denominator = try std.math.mul(i64, left.denominator, right.denominator);
    normalizeScalar(left);
}

fn normalizeScalar(value: *Ope.Scalar) void {
    if (value.numerator == 0) {
        value.denominator = 1;
        value.imaginary_power = 0;
        value.atom_count = 0;
        return;
    }
    if (value.denominator < 0) {
        value.numerator = -value.numerator;
        value.denominator = -value.denominator;
    }
    const divisor: i64 = @intCast(std.math.gcd(absInt(value.numerator), absInt(value.denominator)));
    value.numerator = @divExact(value.numerator, divisor);
    value.denominator = @divExact(value.denominator, divisor);
}

fn absInt(value: i64) u64 {
    if (value == std.math.minInt(i64)) return @as(u64, 1) << 63;
    return if (value < 0) @intCast(-value) else @intCast(value);
}

fn measureStream(comptime case: CaseName, sample_count: usize, comptime Kernel: type, left_terms: []const Ope.Monomial, right_terms: []const Ope.Monomial, projection: Ope.Projection) !Result {
    var samples_storage: [101]u128 = undefined;
    if (sample_count == 0 or sample_count > samples_storage.len) return error.InvalidSampleCount;
    var final_terms: usize = 0;
    var final_hash: u64 = 0;
    for (samples_storage[0..sample_count]) |*sample| {
        var sink = Ope.CountHashSink{};
        const start = nowNs();
        try Kernel.opeProjected(.{ .terms = left_terms }, .{ .terms = right_terms }, projection, &sink);
        sample.* = nowNs() - start;
        final_terms = sink.term_count;
        final_hash = sink.hash;
    }
    std.mem.sort(u128, samples_storage[0..sample_count], {}, struct {
        fn lessThan(_: void, a: u128, b: u128) bool {
            return a < b;
        }
    }.lessThan);
    return .{
        .name = name(case),
        .family = family(case),
        .form = .stream,
        .raw_terms = final_terms,
        .terms = final_terms,
        .hash = final_hash,
        .median_ns = samples_storage[sample_count / 2],
        .samples = sample_count,
        .heap_bytes = 0,
        .collector_capacity = 0,
        .collector_bytes = 0,
    };
}

fn measureCollect(comptime case: CaseName, sample_count: usize, comptime Kernel: type, left_terms: []const Ope.Monomial, right_terms: []const Ope.Monomial, projection: Ope.Projection) !Result {
    var samples_storage: [101]u128 = undefined;
    if (sample_count == 0 or sample_count > samples_storage.len) return error.InvalidSampleCount;
    var final_raw_terms: usize = 0;
    var final_terms: usize = 0;
    var final_hash: u64 = 0;
    const Collector = Ope.FixedCollector(collectCapacity(case));
    const collector_capacity = collectCapacity(case);
    const collector_bytes = @sizeOf(Collector);

        var sink = Collector{};
        const start = nowNs();
        try Kernel.opeProjected(.{ .terms = left_terms }, .{ .terms = right_terms }, projection, &sink);
        sample.* = nowNs() - start;
        sink.finish();
        final_raw_terms = sink.raw_term_count;
        final_terms = sink.term_count;
        final_hash = sink.hash;
    }
    std.mem.sort(u128, samples_storage[0..sample_count], {}, struct {
        fn lessThan(_: void, a: u128, b: u128) bool {
            return a < b;
        }
    }.lessThan);
    return .{
        .name = name(case),
        .family = family(case),
        .form = .collect,
        .raw_terms = final_raw_terms,
        .terms = final_terms,
        .hash = final_hash,
        .median_ns = samples_storage[sample_count / 2],
        .samples = sample_count,
        .heap_bytes = 0,
        .collector_capacity = collector_capacity,
        .collector_bytes = collector_bytes,
    };
}

fn collectCapacity(comptime case: CaseName) usize {

}

    var ts: std.os.linux.timespec = undefined;
    _ = std.os.linux.clock_gettime(.MONOTONIC, &ts);
    return @as(u128, @intCast(ts.sec)) * std.time.ns_per_s + @as(u128, @intCast(ts.nsec));
}

fn maybeRun(comptime case: cases.CaseName, options: Options, form: cases.Form, io: std.Io, file: ?std.Io.File, oracle_json: ?[]const u8, allocator: std.mem.Allocator, emitted: *usize) !void {

fn writeJsonString(io: std.Io, file: ?std.Io.File, value: []const u8) !void {
    try writeBytes(io, file, "\"");
    for (value) |byte| {
        switch (byte) {
            '"' => try writeBytes(io, file, "\\\""),
            '\\' => try writeBytes(io, file, "\\\\"),
            '\n' => try writeBytes(io, file, "\\n"),
            '\r' => try writeBytes(io, file, "\\r"),
            '\t' => try writeBytes(io, file, "\\t"),
            else => try writeBytes(io, file, (&[_]u8{byte})[0..]),
        }
    }
    try writeBytes(io, file, "\"");
}

fn verifyAgainstOracle(json: []const u8, result: cases.Result) !void {
    if (result.form != .collect) return error.InvalidBenchmarkForm;
    const block = findCaseBlock(json, result.name) orelse return error.MissingOracleCase;
    const math_terms = try parseUnsignedField(block, "mathematica_term_count");
    if (math_terms != result.terms) {
        std.debug.print("term-count gate failed case={s} mathematica={} zig={}\n", .{ result.name, math_terms, result.terms });
        return error.OpeOracleTermMismatch;
    }
    if (parseOptionalBoolField(block, "zig_expression_present")) |present| {
        const proportional = parseOptionalBoolField(block, "zig_expression_proportional") orelse false;
        if (present and !proportional) {
            std.debug.print("expression gate failed case={s}: Mathematica did not find proportional equality\n", .{result.name});
            return error.OpeOracleExpressionMismatch;
        }
    }
    const math_s = try parseFloatField(block, "mathematica_median_s");
    const zig_s = @as(f64, @floatFromInt(result.median_ns)) / 1_000_000_000.0;
    if (math_s > 1.0 and zig_s > math_s / 10_000.0) {
        std.debug.print("speed gate failed case={s} mathematica_s={d} zig_s={d}\n", .{ result.name, math_s, zig_s });
        return error.OpeOracleSpeedMismatch;
    }
}

fn findCaseBlock(json: []const u8, name: []const u8) ?[]const u8 {
    var needle_storage: [128]u8 = undefined;
    const needle = std.fmt.bufPrint(&needle_storage, "\"case\":\"{s}\"", .{name}) catch return null;
    const case_index = std.mem.indexOf(u8, json, needle) orelse return null;
    var start = case_index;
    while (start > 0 and json[start] != '{') : (start -= 1) {}
    var end = case_index;
    while (end < json.len and json[end] != '}') : (end += 1) {}
    if (end >= json.len or json[start] != '{') return null;
    return json[start .. end + 1];
}

fn parseUnsignedField(block: []const u8, comptime field: []const u8) !usize {
    const value = fieldValue(block, field) orelse return error.MissingOracleField;
    return std.fmt.parseUnsigned(usize, value, 10);
}

fn parseFloatField(block: []const u8, comptime field: []const u8) !f64 {
    const value = fieldValue(block, field) orelse return error.MissingOracleField;
    return std.fmt.parseFloat(f64, value);
}

fn parseOptionalBoolField(block: []const u8, comptime field: []const u8) ?bool {
    const value = fieldValue(block, field) orelse return null;
    if (std.mem.eql(u8, value, "true")) return true;
    if (std.mem.eql(u8, value, "false")) return false;
    return null;
}

fn fieldValue(block: []const u8, comptime field: []const u8) ?[]const u8 {
    const needle = "\"" ++ field ++ "\":";
    const raw_start = std.mem.indexOf(u8, block, needle) orelse return null;
    var start = raw_start + needle.len;
    while (start < block.len and (block[start] == ' ' or block[start] == '\t')) : (start += 1) {}
    var end = start;
    while (end < block.len and block[end] != ',' and block[end] != '}') : (end += 1) {}
    return std.mem.trim(u8, block[start..end], " \t\n\r");
}

fn parseOptions(raw_args: std.process.Args) !Options {
    var options = Options{};
    var args = std.process.Args.Iterator.init(raw_args);
    _ = args.next();
    while (args.next()) |arg| {
        if (std.mem.eql(u8, arg, "--case")) {
            options.case_filter = args.next() orelse return error.MissingArgument;
        } else if (std.mem.eql(u8, arg, "--family")) {
            options.family_filter = try parseFamily(args.next() orelse return error.MissingArgument);
        } else if (std.mem.eql(u8, arg, "--form")) {
            options.form = try parseForm(args.next() orelse return error.MissingArgument);
        } else if (std.mem.eql(u8, arg, "--emit-json")) {
            options.emit_json = args.next() orelse return error.MissingArgument;
        } else if (std.mem.eql(u8, arg, "--emit-expression")) {
            options.emit_expression = true;
        } else if (std.mem.eql(u8, arg, "--verify-json")) {
            options.verify_json = args.next() orelse return error.MissingArgument;
        } else if (std.mem.eql(u8, arg, "--samples")) {
            options.samples = try std.fmt.parseUnsigned(usize, args.next() orelse return error.MissingArgument, 10);
        } else {
            return error.UnknownArgument;
        }
    }
    return options;
}

fn parseFamily(raw: []const u8) !cases.Family {
    if (std.mem.eql(u8, raw, "bc")) return .bc;
    if (std.mem.eql(u8, raw, "fermion")) return .fermion;
    if (std.mem.eql(u8, raw, "boson")) return .boson;
    return error.UnknownFamily;

fn parseForm(raw: []const u8) !cases.Form {
    if (std.mem.eql(u8, raw, "stream")) return .stream;
    if (std.mem.eql(u8, raw, "collect")) return .collect;
    return error.UnknownForm;
}

        ratio,
        Missing["not-proportional"]
      ]
  ]
];

zigVerificationFields[row_, expr_, zigByCase_] := Module[
  {zigRow, zigExpressionString, zigExpression, exact, factor},
  zigRow = Lookup[zigByCase, row["case"], Missing["zig-case"]];
  If[MissingQ[zigRow], Return[<|"zig_expression_present" -> False|>]];
  zigExpressionString = rowValue[zigRow, "zig_expression"];
  If[MissingQ[zigExpressionString], Return[<|"zig_expression_present" -> False|>]];
  zigExpression = ToExpression[zigExpressionString, InputForm];
  exact = TrueQ[FullSimplify[Expand[expr - zigExpression] === 0]];
  factor = numericOverallFactor[expr, zigExpression];
  <|
    "zig_expression_present" -> True,
    "zig_expression_equal" -> exact,
    "zig_expression_proportional" -> ! MissingQ[factor],
    "zig_expression_factor" -> If[MissingQ[factor], "not-proportional", ToString[InputForm[factor]]]
  |>
];

timedOPEProjectedSeconds[code_, samples_Integer] := Module[{times},
  times = Table[First@AbsoluteTiming[ToExpression[code, InputForm]], {samples}];
  Median[times]
];

allArgs = Join[$CommandLine, $ScriptCommandLine];
firstFlag = FirstPosition[
  allArgs,
  "--case" | "--family" | "--samples" | "--emit-json" | "--verify-zig-json",
  Missing["NotFound"]
];
rawArgs = If[MissingQ[firstFlag], {}, allArgs[[firstFlag[[1]] ;;]]];
opts = parseArgs[rawArgs];
selected = Select[caseRows,
   (opts["case"] === All || #["case"] === opts["case"]) &&
     (opts["family"] === All || #["family"] === opts["family"]) &];
zigRows = If[opts["verify-zig-json"] === None, {}, Import[opts["verify-zig-json"], "JSON"]];
zigByCase = Association[
  Table[rowValue[row, "case"] -> row, {row, zigRows}]
];

results = Table[
   Module[{expr, seconds},
    (* Timing intentionally covers only OPEProjected evaluation.  Expansion,
       hashing, import of Zig output, and proportional equality checks are
       oracle work and are computed after this value is fixed. *)
    seconds = timedOPEProjectedSeconds[row["code"], opts["samples"]];
    expr = ToExpression[row["code"], InputForm];
    Join[
    <|
      "case" -> row["case"],
      "family" -> row["family"],
      "mathematica_term_count" -> termsCount[expr],
      "mathematica_hash" -> canonicalHash[expr],
      "mathematica_median_s" -> seconds,
      "mathematica_timed_operation" -> "OPEProjected only",
      "comparison_timed_in_mathematica_median" -> False,
      "samples" -> opts["samples"],
      "mathematica_version" -> $Version,
      "prototype_path" -> FileNameJoin[{Directory[], "prototype-string-code"}]
    |>,
    If[opts["verify-zig-json"] === None, <||>, zigVerificationFields[row, expr, zigByCase]]
    ]
   ],
   {row, selected}
];

Export[opts["emit-json"], results, "JSON"];
