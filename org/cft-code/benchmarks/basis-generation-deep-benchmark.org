This node contains the reproducible benchmark harnesses used by the benchmark
report.  The Zig side times descriptor construction plus compact streaming.
The Wolfram side uses held benchmark expressions; otherwise Mathematica
evaluates the basis before the timer starts.

#+begin_src zig :tangle ../../../benchmarks/basis_generation_deep_bench.zig :mkdirp yes
const std = @import("std");
const basis = @import("basis-generation");

const max_ticks = 36;
const max_modes = 128;
const max_depth = 48;

const Sink = struct {
    count: usize = 0,
    max_depth_seen: usize = 0,
    hash_mix: u64 = 0,

    pub fn emitBasisState(self: *@This(), candidate: basis.Candidate) !void {
        self.count += 1;
        self.max_depth_seen = @max(self.max_depth_seen, candidate.modes.len);
        self.hash_mix ^= candidate.structuralHash();
    }
};

fn nowNs() u128 {
    var ts: std.os.linux.timespec = undefined;
    _ = std.os.linux.clock_gettime(.MONOTONIC, &ts);
    return @as(u128, @intCast(ts.sec)) * std.time.ns_per_s + @as(u128, @intCast(ts.nsec));
}

fn ContextStorage(comptime quantum_count: usize) type {
    return ContextStorageWithComponents(quantum_count, 0);
}

fn ContextStorageWithComponents(comptime quantum_count: usize, comptime component_count: usize) type {
    return struct {
        word: [max_depth]basis.ModeEntry = undefined,
        component_weights: [component_count]i32 = undefined,
        quantum_values: [quantum_count]i32 = undefined,
        add_min: [basis.Context.reachabilitySlots(max_modes, max_ticks, quantum_count)]i32 = undefined,
        add_max: [basis.Context.reachabilitySlots(max_modes, max_ticks, quantum_count)]i32 = undefined,
        finite_bits: [basis.Context.reachabilitySlots(max_modes, max_ticks, quantum_count)]u64 = undefined,

        fn context(self: *@This()) basis.Context {
            return .{
                .word = &self.word,
                .component_weights = &self.component_weights,
                .quantum_values = &self.quantum_values,
                .add_min = &self.add_min,
                .add_max = &self.add_max,
                .finite_bits = &self.finite_bits,
            };
        }
    };
}

fn printResult(
    comptime name: []const u8,
    comptime category: []const u8,
    weight: i32,
    ghost: i32,
    elapsed_ns: u128,
    scratch_bytes: usize,
    sink: Sink,
) void {
    std.debug.print(
        "zig\t{s}\t{s}\t{}\t{}\t{}\t{}\t{}\t{}\t{}\tok\n",
        .{
            name,
            category,
            weight,
            ghost,
            elapsed_ns,
            sink.count,
            sink.max_depth_seen,
            sink.hash_mix,
            scratch_bytes,
        },
    );
}

fn repairSeedSlices(seeds: []basis.Seed, quantum_storage: []i32, comptime quantum_count: usize) void {
    for (seeds, 0..) |*seed, index| {
        seed.quantum_values = quantum_storage[index * quantum_count .. (index + 1) * quantum_count];
    }
}

fn runFreeBosonExact(comptime weight: u32) !void {
    var mode_storage: [max_modes]basis.Mode = undefined;
    var storage = ContextStorage(0){};
    var sink = Sink{};

    const start = nowNs();
    const families = [_]basis.OscillatorFamily{basis.Preset.freeBosonFamily(1, 0, 1)};
    const modes = try basis.buildOscillatorModes(&families, weight, &mode_storage);
    var context = storage.context();
    try basis.stream(.{ .modes = modes }, .{
        .weight = .{ .exact = weight },
        .max_word_length = max_depth,
    }, &context, &sink);
    printResult("free-boson-exact-" ++ comptime std.fmt.comptimePrint("{}", .{weight}), "compact-free-boson", weight, 0, nowNs() - start, @sizeOf(@TypeOf(storage)), sink);
}

fn runBcCumulative(comptime weight: u32, comptime ghost: i32) !void {
    var mode_storage: [max_modes]basis.Mode = undefined;
    var seed_storage: [4]basis.Seed = undefined;
    var seed_quantum: [4]i32 = undefined;
    var seed_options_storage: [2]basis.SeedOption = undefined;
    var storage = ContextStorageWithComponents(basis.Preset.ghost_schema.len, 2){};
    var sink = Sink{};

    const start = nowNs();
    const seed_families = [_]basis.SeedFamily{basis.Preset.bcSeedFamily(1, 0)};
    const seed_options = try basis.buildSeedOptions(&seed_families, &seed_options_storage);
    const seeds = try basis.buildFermionicSeedSubsets(seed_options, basis.Preset.ghost_schema.len, &seed_storage, &seed_quantum);
    repairSeedSlices(seed_storage[0..seeds.len], &seed_quantum, basis.Preset.ghost_schema.len);
    const mode_families = basis.Preset.bcOscillatorFamilies(2, 3, 0);
    const modes = try basis.buildOscillatorModes(&mode_families, weight + 1, &mode_storage);
    var context = storage.context();
    try basis.stream(.{
        .quantum_schema = &basis.Preset.ghost_schema,
        .modes = modes,
        .seeds = seeds,
    }, .{
        .weight = .{ .max = weight },
        .quantum_filters = &.{.{ .slot = 0, .value = ghost }},
        .max_word_length = max_depth,
    }, &context, &sink);
    printResult("bc-cumulative-" ++ comptime std.fmt.comptimePrint("{}-g{}", .{ weight, ghost }), "compact-bc", weight, ghost, nowNs() - start, @sizeOf(@TypeOf(storage)), sink);
}

fn runBosonBcExact(comptime weight: u32, comptime ghost: i32) !void {
    var mode_storage: [max_modes]basis.Mode = undefined;
    var seed_storage: [4]basis.Seed = undefined;
    var seed_quantum: [4]i32 = undefined;
    var seed_options_storage: [2]basis.SeedOption = undefined;
    var storage = ContextStorageWithComponents(basis.Preset.ghost_schema.len, 2){};
    var sink = Sink{};

    const start = nowNs();
    const seed_families = [_]basis.SeedFamily{basis.Preset.bcSeedFamily(1, 0)};
    const seed_options = try basis.buildSeedOptions(&seed_families, &seed_options_storage);
    const seeds = try basis.buildFermionicSeedSubsets(seed_options, basis.Preset.ghost_schema.len, &seed_storage, &seed_quantum);
    repairSeedSlices(seed_storage[0..seeds.len], &seed_quantum, basis.Preset.ghost_schema.len);

    const matter_families = [_]basis.OscillatorFamily{basis.Preset.freeBosonFamily(4, 0, 1)};
    const matter_modes = try basis.buildOscillatorModes(&matter_families, weight + 1, &mode_storage);
    const bc_families = basis.Preset.bcOscillatorFamilies(2, 3, 0);
    const bc_modes = try basis.buildOscillatorModes(&bc_families, weight + 1, mode_storage[matter_modes.len..]);
    const modes = mode_storage[0 .. matter_modes.len + bc_modes.len];

    var context = storage.context();
    try basis.stream(.{
        .quantum_schema = &basis.Preset.ghost_schema,
        .modes = modes,
        .seeds = seeds,
    }, .{
        .weight = .{ .exact = weight },
        .quantum_filters = &.{.{ .slot = 0, .value = ghost }},
        .max_word_length = max_depth,
    }, &context, &sink);
    printResult("boson-bc-exact-" ++ comptime std.fmt.comptimePrint("{}-g{}", .{ weight, ghost }), "compact-boson-bc", weight, ghost, nowNs() - start, @sizeOf(@TypeOf(storage)), sink);
}

fn buildClosedBcSeeds(seeds: []basis.Seed, quantum_storage: []i32, component_weight_storage: []i32) ![]const basis.Seed {
    if (seeds.len < 16 or quantum_storage.len < 16 or component_weight_storage.len < 32) return error.ContextTooSmall;
    const option_weights = [_]i32{ -1, 0, -1, 0 };
    const option_components = [_]usize{ 0, 0, 1, 1 };

    var mask: usize = 0;
    while (mask < 16) : (mask += 1) {
        const quantum = quantum_storage[mask .. mask + 1];
        const component_weights = component_weight_storage[mask * 2 .. mask * 2 + 2];
        quantum[0] = 0;
        component_weights[0] = 0;
        component_weights[1] = 0;

        var weight: i32 = 0;
        for (option_weights, 0..) |option_weight, bit| {
            if ((mask & (@as(usize, 1) << @intCast(bit))) == 0) continue;
            weight += option_weight;
            quantum[0] += 1;
            component_weights[option_components[bit]] += option_weight;
        }

        seeds[mask] = .{
            .id = @intCast(mask),
            .weight_ticks = weight,
            .component_weight_ticks = component_weights,
            .quantum_values = quantum,
        };
    }
    return seeds[0..16];
}

fn appendModes(families: []const basis.OscillatorFamily, comptime weight: u32, modes: []basis.Mode, offset: *usize) !void {
    const written = try basis.buildOscillatorModes(families, weight + 1, modes[offset.*..]);
    offset.* += written.len;
}

fn runClosedBosonic(comptime weight: u32, comptime level_matched: bool) !void {
    var mode_storage: [max_modes]basis.Mode = undefined;
    var seed_storage: [16]basis.Seed = undefined;
    var seed_quantum: [16]i32 = undefined;
    var seed_component_weights: [32]i32 = undefined;
    var storage = ContextStorageWithComponents(basis.Preset.ghost_schema.len, 2){};
    var sink = Sink{};

    const start = nowNs();
    const seeds = try buildClosedBcSeeds(&seed_storage, &seed_quantum, &seed_component_weights);
    var mode_count: usize = 0;
    const left_matter = [_]basis.OscillatorFamily{basis.Preset.freeBosonFamily(4, 0, 1)};
    const right_matter = [_]basis.OscillatorFamily{basis.Preset.freeBosonFamily(5, 1, 1)};
    const left_bc = basis.Preset.bcOscillatorFamilies(2, 3, 0);
    const right_bc = basis.Preset.bcOscillatorFamilies(6, 7, 1);
    try appendModes(&left_matter, weight, &mode_storage, &mode_count);
    try appendModes(&left_bc, weight, &mode_storage, &mode_count);
    try appendModes(&right_matter, weight, &mode_storage, &mode_count);
    try appendModes(&right_bc, weight, &mode_storage, &mode_count);

    var context = storage.context();
    var query = basis.Query{
        .weight = .{ .exact = weight },
        .quantum_filters = &.{.{ .slot = 0, .value = 0 }},
        .max_word_length = max_depth,
    };
    if (level_matched) query.level_match = .{};

    try basis.stream(.{
        .quantum_schema = &basis.Preset.ghost_schema,
        .modes = mode_storage[0..mode_count],
        .seeds = seeds,
    }, query, &context, &sink);

    const category = if (level_matched) "compact-closed-levelmatched" else "compact-closed-all-splits";
    const name = if (level_matched)
        "closed-levelmatched-" ++ comptime std.fmt.comptimePrint("{}-g0", .{weight})
    else
        "closed-all-splits-" ++ comptime std.fmt.comptimePrint("{}-g0", .{weight});
    printResult(name, category, weight, 0, nowNs() - start, @sizeOf(@TypeOf(storage)), sink);
}

pub fn main() !void {
    std.debug.print("engine\tname\tcategory\tweight\tghost\tns\tcount\tmax_depth\thash_or_bytes\tscratch_bytes\tstatus\n", .{});

    inline for (.{ 8, 12, 16, 20, 24, 28 }) |weight| {
        try runFreeBosonExact(weight);
    }

    inline for (.{ 8, 10, 12, 14, 16, 20, 24, 28 }) |weight| {
        try runBosonBcExact(weight, 0);
        try runBosonBcExact(weight, 1);
    }

    inline for (.{ 8, 12, 16, 20, 24 }) |weight| {
        try runBcCumulative(weight, 0);
        try runBcCumulative(weight, 1);
    }

    inline for (.{ 3, 5, 7, 9, 11, 13 }) |half_tick_weight| {
        try runNsFermionOdd(half_tick_weight);
    }

    inline for (.{ 6, 8, 10, 12 }) |weight| {
        try runClosedBosonic(weight, true);
    }
    inline for (.{ 4, 6, 8 }) |weight| {
        try runClosedBosonic(weight, false);
    }
}
