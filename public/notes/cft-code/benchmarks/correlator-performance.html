Correlator Performance Benchmarks

This benchmark times the streamed Wick correlator on representative free-boson, free-fermion, ghost, and exponential workloads. It reports medians, not best runs. The event sink counts the fully expanded stream; the branch sink counts the compact branch path and its stack state size.

The event lines describe the expanded primitive event stream: a Wick term boundary, scalar/coordinate/tensor/action factors, residual operator references, and the zero-mode base-case factors. The branch lines describe one compact accumulated branch object per Wick/base-case branch, while zero modes remain separate events. fallback_terms is nonzero only when a case cannot use the compact branch path.

The runner enforces a 5% median regression gate for every event and branch case against the recorded baselines below. Cases with very small baselines run multiple correlators per sample and report the per-call median, keeping the speed gate strict while reducing timer noise. Exponential and mixed cases also check primitive-term, tensor-factor, residual, zero-mode, and base-case counts so semantic drift fails the benchmark instead of only changing printed numbers. 

const std = @import("std");
const cft = @import("cft-code");

const max_sample_count = 101;

const MedianGate = struct {
    baseline_ns: u128,
    limit_num: u128,
    limit_den: u128,
};

const ExpectedCounts = struct {
    primitive_terms: usize,
    tensors: usize,
    residuals: usize,
    zero_modes: usize,
    base_cases: usize,
};

fn medianGate(comptime baseline_ns: u128) MedianGate {
    return .{ .baseline_ns = baseline_ns, .limit_num = 21, .limit_den = 20 };
}

fn measurementBatchCount(baseline_ns: u128) usize {
    if (baseline_ns < 20_000) return 32;
    if (baseline_ns < 50_000) return 8;
    return 1;
}

fn nowNs() u128 {
    var ts: std.os.linux.timespec = undefined;
    _ = std.os.linux.clock_gettime(.MONOTONIC, &ts);
    return @as(u128, @intCast(ts.sec)) * std.time.ns_per_s + @as(u128, @intCast(ts.nsec));
}

const EventSink = struct {
    wick_terms: usize = 0,
    scalars: usize = 0,
    coordinates: usize = 0,
    tensors: usize = 0,
    actions: usize = 0,
    residuals: usize = 0,
    zero_modes: usize = 0,
    base_cases: usize = 0,

    pub fn emitWickTermStart(self: *@This(), _: anytype) !void {
        self.wick_terms += 1;
    }

    pub fn emitWickScalar(self: *@This(), _: anytype) !void {
        self.scalars += 1;
    }

    pub fn emitWickCoordinate(self: *@This(), _: anytype) !void {
        self.coordinates += 1;
    }

    pub fn emitWickTensor(self: *@This(), _: anytype) !void {
        self.tensors += 1;
    }

    pub fn emitWickAction(self: *@This(), _: anytype) !void {
        self.actions += 1;
    }

    pub fn emitWickResidualOperator(self: *@This(), _: anytype) !void {
        self.residuals += 1;
    }

    pub fn emitWickTermEnd(_: *@This()) !void {}

    pub fn emitZeroModeFactor(self: *@This(), _: anytype) !void {
        self.zero_modes += 1;
    }

    pub fn emitZeroModeBaseEnd(self: *@This()) !void {
        self.base_cases += 1;
    }
};

const BranchSink = struct {
    branch_terms: usize = 0,
    fallback_terms: usize = 0,
    primitive_terms: usize = 0,
    scalars: usize = 0,
    coordinates: usize = 0,
    tensors: usize = 0,
    actions: usize = 0,
    residuals: usize = 0,
    zero_modes: usize = 0,
    base_cases: usize = 0,
    branch_state_bytes: usize = 0,

    pub fn emitWickBranchTerm(self: *@This(), branch: anytype) !void {
        self.branch_terms += 1;
        self.primitive_terms += branch.primitiveTermCount();
        self.scalars += branch.scalarFactorCount();
        self.coordinates += branch.coordinateFactorCount();
        self.tensors += branch.tensorFactorCount();
        self.actions += branch.actionFactorCount();
        self.residuals += branch.residualFactorCount();
        const bytes = branchStateBytes(branch);
        if (bytes > self.branch_state_bytes) self.branch_state_bytes = bytes;
    }

    pub fn emitWickTermStart(self: *@This(), _: anytype) !void {
        self.fallback_terms += 1;
        self.primitive_terms += 1;
    }

    pub fn emitWickScalar(self: *@This(), _: anytype) !void {
        self.scalars += 1;
    }

    pub fn emitWickCoordinate(self: *@This(), _: anytype) !void {
        self.coordinates += 1;
    }

    pub fn emitWickTensor(self: *@This(), _: anytype) !void {
        self.tensors += 1;
    }

    pub fn emitWickAction(self: *@This(), _: anytype) !void {
        self.actions += 1;
    }

    pub fn emitWickResidualOperator(self: *@This(), _: anytype) !void {
        self.residuals += 1;
    }

    pub fn emitWickTermEnd(_: *@This()) !void {}

    pub fn emitZeroModeFactor(self: *@This(), _: anytype) !void {
        self.zero_modes += 1;
    }

    pub fn emitZeroModeBaseEnd(self: *@This()) !void {
        self.base_cases += 1;
    }
};

fn branchStateBytes(branch: anytype) usize {
    return switch (@typeInfo(@TypeOf(branch))) {
        .pointer => |ptr| @sizeOf(ptr.child),
        else => @sizeOf(@TypeOf(branch)),
    };
}

fn median(samples: []u128) u128 {
    std.mem.sort(u128, samples, {}, struct {
        fn lessThan(_: void, left: u128, right: u128) bool {
            return left < right;
        }
    }.lessThan);
    return samples[samples.len / 2];
}

fn requireSampleCount(comptime sample_count: usize) void {
    if (sample_count == 0 or sample_count > max_sample_count or (sample_count & 1) == 0) {
        @compileError("benchmark sample counts must be odd and <= max_sample_count");
    }
}

fn checkEventCounts(comptime name: []const u8, result: EventSink, counts: ExpectedCounts, batch_count: usize) !void {
    if (result.wick_terms != counts.primitive_terms * batch_count or
        result.tensors != counts.tensors * batch_count or
        result.residuals != counts.residuals * batch_count or
        result.zero_modes != counts.zero_modes * batch_count or
        result.base_cases != counts.base_cases * batch_count)
    {
        std.debug.print(
            "event count regression case={s} primitive={} tensors={} residuals={} zero_modes={} base_cases={}\n",
            .{ name, result.wick_terms, result.tensors, result.residuals, result.zero_modes, result.base_cases },
        );
        return error.BenchmarkCountRegression;
    }
}

fn checkBranchCounts(comptime name: []const u8, result: BranchSink, counts: ExpectedCounts, batch_count: usize) !void {
    if (result.primitive_terms != counts.primitive_terms * batch_count or
        result.tensors != counts.tensors * batch_count or
        result.residuals != counts.residuals * batch_count or
        result.zero_modes != counts.zero_modes * batch_count or
        result.base_cases != counts.base_cases * batch_count)
    {
        std.debug.print(
            "branch count regression case={s} primitive={} tensors={} residuals={} zero_modes={} base_cases={}\n",
            .{ name, result.primitive_terms, result.tensors, result.residuals, result.zero_modes, result.base_cases },
        );
        return error.BenchmarkCountRegression;
    }
}

fn checkMedianGate(comptime form: []const u8, comptime name: []const u8, median_ns: u128, gate: MedianGate) !void {
    if (median_ns * gate.limit_den > gate.baseline_ns * gate.limit_num) {
        std.debug.print(
            "{s} speed regression case={s} median_ns={} baseline_ns={} limit={}/{}\n",
            .{ form, name, median_ns, gate.baseline_ns, gate.limit_num, gate.limit_den },
        );
        return error.BenchmarkSpeedRegression;
    }
}

fn benchEvent(comptime name: []const u8, comptime sample_count: usize, theory: anytype, config: anytype, ops: anytype, gate: MedianGate, expected: ExpectedCounts) !void {
    comptime requireSampleCount(sample_count);

    const batch_count = measurementBatchCount(gate.baseline_ns);
    var samples: [max_sample_count]u128 = undefined;
    var result = EventSink{};
    for (samples[0..sample_count]) |*sample| {
        var sink = EventSink{};
        const start = nowNs();
        for (0..batch_count) |_| {
            try theory.correlator(config, ops, &sink);
        }
        sample.* = (nowNs() - start) / batch_count;
        result = sink;
    }

    const median_ns = median(samples[0..sample_count]);
    try checkMedianGate("event", name, median_ns, gate);
    try checkEventCounts(name, result, expected, batch_count);
    std.debug.print(
        "event case={s} samples={} batch={} median_ns={} wick_terms={} scalars={} coordinates={} tensors={} actions={} residuals={} zero_modes={} base_cases={}\n",
        .{ name, sample_count, batch_count, median_ns, result.wick_terms, result.scalars, result.coordinates, result.tensors, result.actions, result.residuals, result.zero_modes, result.base_cases },
    );
}

fn benchBranch(comptime name: []const u8, comptime sample_count: usize, theory: anytype, config: anytype, ops: anytype, gate: MedianGate, expected: ExpectedCounts) !void {
    comptime requireSampleCount(sample_count);

    const batch_count = measurementBatchCount(gate.baseline_ns);
    var samples: [max_sample_count]u128 = undefined;
    var result = BranchSink{};
    for (samples[0..sample_count]) |*sample| {
        var sink = BranchSink{};
        const start = nowNs();
        for (0..batch_count) |_| {
            try theory.correlator(config, ops, &sink);
        }
        sample.* = (nowNs() - start) / batch_count;
        result = sink;
    }

    const median_ns = median(samples[0..sample_count]);
    try checkMedianGate("branch", name, median_ns, gate);
    try checkBranchCounts(name, result, expected, batch_count);
    std.debug.print(
        "branch case={s} samples={} batch={} median_ns={} branch_terms={} fallback_terms={} primitive_terms={} scalars={} coordinates={} tensors={} actions={} residuals={} zero_modes={} base_cases={} branch_state_bytes={}\n",
        .{ name, sample_count, batch_count, median_ns, result.branch_terms, result.fallback_terms, result.primitive_terms, result.scalars, result.coordinates, result.tensors, result.actions, result.residuals, result.zero_modes, result.base_cases, result.branch_state_bytes },
    );
}

fn benchFreeBosonDx(comptime n: usize, comptime sample_count: usize, allocator: std.mem.Allocator, event_gate: MedianGate, branch_gate: MedianGate, expected: ExpectedCounts) !void {
    const X = cft.FreeBoson.make(.{ .dimension = 10 });

    var type_local = try X.local(allocator);
    const probe_mu = try type_local.index("probe_mu");
    const probe_z = try type_local.coord("probe_z");
    var ops_buf: [n]@TypeOf(try X.op.dX(&type_local, probe_mu, 0, probe_z)) = undefined;
    type_local.deinit();

    var local = try X.local(allocator);
    defer local.deinit();

    const mu = try local.index("mu");
    inline for (0..n) |i| {
        const z = try local.coord(std.fmt.comptimePrint("z{}", .{i}));
        ops_buf[i] = try X.op.dX(&local, mu, 0, z);
    }
    const ops = try local.ops(ops_buf);
    const name = std.fmt.comptimePrint("dx{}", .{n});

    try benchEvent(name, sample_count, X, &X.config.sphere, ops, event_gate, expected);
    try benchBranch(name, sample_count, X, &X.config.sphere, ops, branch_gate, expected);
}

fn benchFreeFermionPsi(comptime n: usize, comptime sample_count: usize, allocator: std.mem.Allocator, event_gate: MedianGate, branch_gate: MedianGate, expected: ExpectedCounts) !void {
    const Psi = cft.FreeFermion.sphere(.{ .dimension = 10, .include_antiholomorphic_copy = false });

    var type_local = try Psi.local(allocator);
    const probe_mu = try type_local.index("probe_mu");
    const probe_z = try type_local.coord("probe_z");
    var ops_buf: [n]@TypeOf(try Psi.op.psi(&type_local, probe_mu, 0, probe_z)) = undefined;
    type_local.deinit();

    var local = try Psi.local(allocator);
    defer local.deinit();

    inline for (0..n) |i| {
        const mu = try local.index(std.fmt.comptimePrint("mu{}", .{i}));
        const z = try local.coord(std.fmt.comptimePrint("psi{}", .{i}));
        ops_buf[i] = try Psi.op.psi(&local, mu, 0, z);
    }
    const ops = try local.ops(ops_buf);
    const name = std.fmt.comptimePrint("psi{}", .{n});

    if (n >= 2) {
        const TensorGuard = struct {
            metric_count: usize = 0,

            pub fn emitWickTermStart(_: *@This(), _: anytype) !void {}
            pub fn emitWickScalar(_: *@This(), _: anytype) !void {}
            pub fn emitWickCoordinate(_: *@This(), _: anytype) !void {}

            pub fn emitWickTensor(self: *@This(), factor: anytype) !void {
                switch (factor) {
                    .metric => |metric| {
                        switch (metric.left) {
                            .symbol => {},
                            else => return error.UnexpectedPsiTensorLabel,
                        }
                        switch (metric.right) {
                            .symbol => {},
                            else => return error.UnexpectedPsiTensorLabel,
                        }
                        self.metric_count += 1;
                    },
                    else => return error.UnexpectedPsiTensorFactor,
                }
            }

            pub fn emitWickAction(_: *@This(), _: anytype) !void {}
            pub fn emitWickTermEnd(_: *@This()) !void {}
            pub fn emitZeroModeFactor(_: *@This(), _: anytype) !void {}
            pub fn emitZeroModeBaseEnd(_: *@This()) !void {}
        };

        var guard = TensorGuard{};
        try Psi.correlator(&Psi.config.sphere, ops, &guard);
        if (guard.metric_count != expected.tensors) return error.BenchmarkCountRegression;
    }

    try benchEvent(name, sample_count, Psi, &Psi.config.sphere, ops, event_gate, expected);
    try benchBranch(name, sample_count, Psi, &Psi.config.sphere, ops, branch_gate, expected);
}
