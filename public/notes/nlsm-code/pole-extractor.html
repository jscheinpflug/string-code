NLSM Kernel Evaluator

This note widens the old pole-only stage into a real kernel evaluator. The reduced branch rows emitted by the contraction side are matched against an explicit analytic rule table. The output is no longer only the UV pole. Instead we emit a streamed analytic row that can carry:

  • Laurent poles in \(\epsilon\);
  • local finite pieces;
  • nonlocal finite pieces;
  • explicit powers of \(\log \mu\).

That split matters physically. For beta functions, only the local pole projection is needed. For more general quantization problems, such as perturbed two-point functions and spectrum shifts, the finite and nonlocal pieces must survive as first-class output.

The current evaluator is still a bootstrap rule table, but it no longer pretends that every two-loop branch with two external derivative legs is the same object. The reduced signature now keeps:

  • loop order;
  • vertex count;
  • pair count;
  • separate \(\partial X_0\) and \(\bar\partial X_0\) background-leg counts;
  • primitive pair-kernel histogram;
  • total derivative order on left-moving fermion pairs;
  • total derivative order on right-moving fermion pairs;
  • propagator powers and numerator rank.

This is enough to assign a distinct analytic signature to the first mixed supersymmetric two-loop branch built from one left-moving and one right-moving connection vertex, even though the current bootstrap rule table still maps it to the same placeholder Laurent template as the other nested-bubble examples. 

const std = @import("std");
const geometry = @import("local_geometry_reducer.zig");
const scheme = @import("scheme.zig");
const counterterms = @import("counterterms.zig");

/// KernelTermRow is one reduced local branch before analytic kernel evaluation.
pub const KernelTermRow = struct {
    scheme: scheme.DimRegMS,
    local_operator: counterterms.LocalCountertermKind,
    target_flavor: geometry.GeometryFlavor = .generic_riemannian,
    alpha_prime_power: i16 = 0,
    combinatorial: counterterms.Rational = .{ .numerator = 1, .denominator = 1 },
    kernel: counterterms.KernelSignature,
    term: geometry.TensorTerm,
};

/// LocalityClass records whether one evaluated contribution is local, nonlocal, or scaleless.
pub const LocalityClass = enum(u8) {
    local,
    non_local,
    scaleless,
};

/// KernelEvaluationContribution stores one Laurent or finite contribution of a matched kernel family.
pub const KernelEvaluationContribution = struct {
    locality: LocalityClass = .local,
    pole_order: u8 = 0,
    log_mu_power: u8 = 0,
    coefficient: counterterms.Rational,
};

/// KernelEvaluationRule maps one reduced kernel signature to explicit evaluated contributions.
pub const KernelEvaluationRule = struct {
    signature: counterterms.KernelSignature,
    contributions: []const KernelEvaluationContribution = &.{},
};

/// KernelEvaluationRow is one analytic contribution emitted after kernel evaluation.
pub const KernelEvaluationRow = struct {
    scheme: scheme.DimRegMS,
    local_operator: counterterms.LocalCountertermKind,
    target_flavor: geometry.GeometryFlavor = .generic_riemannian,
    loop_order: u8,
    alpha_prime_power: i16 = 0,
    locality: LocalityClass,
    pole_order: u8 = 0,
    log_mu_power: u8 = 0,
    coefficient: counterterms.Rational,
    kernel: counterterms.KernelSignature,
    term: geometry.TensorTerm,

    /// isPole reports whether the row carries a nonzero Laurent pole.
    pub fn isPole(self: KernelEvaluationRow) bool {
        return self.pole_order != 0;
    }

    /// isFinite reports whether the row contributes at epsilon^0.
    pub fn isFinite(self: KernelEvaluationRow) bool {
        return self.pole_order == 0;
    }
};

/// KernelEvaluationSummary reports how the kernel rows were classified and emitted.
pub const KernelEvaluationSummary = struct {
    emitted_count: usize = 0,
    emitted_local_count: u32 = 0,
    emitted_non_local_count: u32 = 0,
    emitted_finite_count: u32 = 0,
    scaleless_count: u32 = 0,
    unmatched_count: u32 = 0,
};

/// PoleProjectionSummary reports how evaluated rows were projected to local pole rows.
pub const PoleProjectionSummary = struct {
    emitted_pole_count: usize = 0,
    skipped_finite_count: u32 = 0,
    skipped_non_local_count: u32 = 0,
    skipped_scaleless_count: u32 = 0,
};

fn sameSignature(left: counterterms.KernelSignature, right: counterterms.KernelSignature) bool {
    return left.family == right.family and
        left.loop_order == right.loop_order and
        left.propagator_len == right.propagator_len and
        left.numerator_rank == right.numerator_rank and
        left.external_derivative_order == right.external_derivative_order and
        std.mem.eql(u8, left.primitive_pair_histogram[0..], right.primitive_pair_histogram[0..]) and
        std.mem.eql(u8, left.propagator_powers[0..], right.propagator_powers[0..]);
}

fn findRule(rules: []const KernelEvaluationRule, signature: counterterms.KernelSignature) ?KernelEvaluationRule {
    for (rules) |rule| {
        if (sameSignature(rule.signature, signature)) return rule;
    }
    return null;
}

/// evaluateKernelRows matches reduced kernel signatures to explicit Laurent/finite evaluation rules.
pub fn evaluateKernelRows(rows: []const KernelTermRow, rules: []const KernelEvaluationRule, output: []KernelEvaluationRow) !KernelEvaluationSummary {
    var summary = KernelEvaluationSummary{};

    for (rows) |row| {
        if (row.kernel.isScaleless()) {
            summary.scaleless_count += 1;
            continue;
        }

        const rule = findRule(rules, row.kernel) orelse {
            summary.unmatched_count += 1;
            continue;
        };

        if (rule.contributions.len == 0) {
            summary.scaleless_count += 1;
            continue;
        }

        for (rule.contributions) |contribution| {
            if (contribution.locality == .scaleless) {
                summary.scaleless_count += 1;
                continue;
            }
            if (summary.emitted_count == output.len) return error.OutputTooSmall;

            output[summary.emitted_count] = .{
                .scheme = row.scheme,
                .local_operator = row.local_operator,
                .target_flavor = row.target_flavor,
                .loop_order = row.kernel.loop_order,
                .alpha_prime_power = row.alpha_prime_power,
                .locality = contribution.locality,
                .pole_order = contribution.pole_order,
                .log_mu_power = contribution.log_mu_power,
                .coefficient = try row.combinatorial.mul(contribution.coefficient),
                .kernel = row.kernel,
                .term = row.term,
            };
            summary.emitted_count += 1;
            switch (contribution.locality) {
                .local => summary.emitted_local_count += 1,
                .non_local => summary.emitted_non_local_count += 1,
                .scaleless => unreachable,
            }
            if (contribution.pole_order == 0) summary.emitted_finite_count += 1;
        }
    }

    return summary;
}

/// extractPoleRows projects the local pole part of evaluated kernel rows.
pub fn extractPoleRows(rows: []const KernelEvaluationRow, output: []counterterms.PoleRow) !PoleProjectionSummary {
    var summary = PoleProjectionSummary{};

    for (rows) |row| {
        switch (row.locality) {
            .scaleless => {
                summary.skipped_scaleless_count += 1;
                continue;
            },
            .non_local => {
                summary.skipped_non_local_count += 1;
                continue;
            },
            .local => {},
        }

        if (!row.isPole()) {
            summary.skipped_finite_count += 1;
            continue;
        }

        if (summary.emitted_pole_count == output.len) return error.OutputTooSmall;
        output[summary.emitted_pole_count] = .{
            .scheme = row.scheme,
            .local_operator = row.local_operator,
            .target_flavor = row.target_flavor,
            .loop_order = row.loop_order,
            .pole_order = row.pole_order,
            .alpha_prime_power = row.alpha_prime_power,
            .residue = row.coefficient,
            .kernel = row.kernel,
            .term = row.term,
        };
        summary.emitted_pole_count += 1;
    }

    return summary;
}

/// stringbookBootstrapEvaluations returns the first explicit evaluation table used for low-loop checks.
pub fn stringbookBootstrapEvaluations() []const KernelEvaluationRule {
    return &bootstrap_rules;
}

test "kernel evaluator emits pole, finite, and nonlocal rows from matched signatures" {
    const testing = std.testing;
    const slots = [_]geometry.TensorSlot{
        .{ .id = 1, .sort = .real_tangent },
        .{ .id = 2, .sort = .real_tangent },
        .{ .id = 3, .sort = .real_tangent },
        .{ .id = 4, .sort = .real_tangent },
    };
    const atoms = [_]geometry.TensorAtom{.{ .kind = .riemann, .slots = &slots }};
    const rows = [_]KernelTermRow{.{
        .scheme = scheme.stringbookMS(),
        .local_operator = .metric_beta,
        .alpha_prime_power = 1,
        .kernel = .{
            .family = .bubble,
            .loop_order = 1,
            .propagator_len = 2,
            .propagator_powers = [_]u8{ 1, 1, 0, 0, 0, 0, 0, 0 },
            .external_derivative_order = 2,
        },
        .term = .{ .atoms = &atoms },
    }};

    var output: [4]KernelEvaluationRow = undefined;
    const summary = try evaluateKernelRows(&rows, stringbookBootstrapEvaluations(), &output);
    try testing.expectEqual(@as(usize, 3), summary.emitted_count);
    try testing.expectEqual(@as(u32, 2), summary.emitted_local_count);
    try testing.expectEqual(@as(u32, 1), summary.emitted_non_local_count);
    try testing.expectEqual(@as(u32, 2), summary.emitted_finite_count);
    try testing.expectEqual(@as(u8, 1), output[0].pole_order);
    try testing.expectEqual(@as(u8, 1), output[2].log_mu_power);
    try testing.expectEqual(LocalityClass.non_local, output[2].locality);
}

test "pole extractor preserves calabi-yau target flavor for later quotients" {
    const testing = std.testing;
    const slots = [_]geometry.TensorSlot{
        .{ .id = 1, .sort = .holomorphic_tangent },
        .{ .id = 2, .sort = .antiholomorphic_tangent },
    };
    const atoms = [_]geometry.TensorAtom{.{ .kind = .ricci, .slots = &slots }};
    const rows = [_]KernelTermRow{.{
        .scheme = scheme.stringbookMS(),
        .local_operator = .kahler_potential_beta,
        .target_flavor = .calabi_yau,
        .alpha_prime_power = 3,
        .kernel = .{
            .family = .nested_bubble,
            .loop_order = 2,
            .propagator_len = 4,
            .propagator_powers = [_]u8{ 1, 1, 1, 1, 0, 0, 0, 0 },
            .external_derivative_order = 2,
        },
        .term = .{ .atoms = &atoms },
    }};

    var evaluated: [4]KernelEvaluationRow = undefined;
    const evaluation_summary = try evaluateKernelRows(&rows, stringbookBootstrapEvaluations(), &evaluated);
    try testing.expectEqual(@as(usize, 3), evaluation_summary.emitted_count);
    try testing.expectEqual(geometry.GeometryFlavor.calabi_yau, evaluated[0].target_flavor);

    var poles: [4]counterterms.PoleRow = undefined;
    const pole_summary = try extractPoleRows(evaluated[0..evaluation_summary.emitted_count], &poles);
    try testing.expectEqual(@as(usize, 2), pole_summary.emitted_pole_count);
    try testing.expectEqual(counterterms.LocalCountertermKind.kahler_potential_beta, poles[0].local_operator);
    try testing.expectEqual(geometry.GeometryFlavor.calabi_yau, poles[0].target_flavor);
}

    const rows = [_]KernelEvaluationRow{
        .{
            .scheme = scheme.stringbookMS(),
            .local_operator = .metric_beta,

            .locality = .local,

    const testing = std.testing;

        .{
            .scheme = scheme.stringbookMS(),

            .kernel = .{ .family = .scaleless_tadpole, .loop_order = 1 },
            .term = .{},
        },
        .{
            .scheme = scheme.stringbookMS(),
            .local_operator = .metric_beta,
            .kernel = .{ .family = .ladder, .loop_order = 4 },
            .term = .{},
        },
    };

    var output: [1]KernelEvaluationRow = undefined;
    const summary = try evaluateKernelRows(&rows, stringbookBootstrapEvaluations(), &output);
    try testing.expectEqual(@as(u32, 1), summary.scaleless_count);

}
